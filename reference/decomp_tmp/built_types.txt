	private static bool RelationMatch(string relCn, string relation)
	{

		{
		case "亲族":
			switch (relCn)
			{
			default:
				return relCn == "私生子女";
			case "父母":
			case "子女":
			case "兄弟姐妹":
			case "义父母":
			case "义子女":
				return true;
			}
		case "道侣":
			if (!(relCn == "道侣"))
			{
				return relCn == "情人";
			}
			return true;
		case "师徒":
			if (!(relCn == "师傅"))
			{
				return relCn == "徒弟";
			}
			return true;
		case "结义":
			return relCn == "结义";
		case "好感":
		case "好友":
			return true;
		case "陌生":
			if (!(relCn == "None"))
			{
				return relCn == "";
			}
			return true;
		default:
			return true;
		}
	}

//==================== TYPE AgentLoopVerify.Bootstrap ====================
using System;
using Il2CppSystem;
using UnityEngine;

namespace AgentLoopVerify;

public static class Bootstrap
{
	private static TimerCoroutine _cor;

	private static VerifyRunner _runner;

	private static bool _announced;

	public static void Start()
	{
		try
		{
			if (_cor != null)
			{
				try
				{
					g.timer.Stop(_cor);
				}
				catch
				{
				}
				_cor = null;
			}
			_announced = false;
			_runner = new VerifyRunner();
			_cor = g.timer.Frame(Action.op_Implicit((Action)OnUpdate), 1, true);
			Log("已启动：等待世界就绪，按 F8 逐项执行工具验证");
		}
		catch (Exception ex)
		{
			Log("Start 失败: " + ex);
		}
	}

	private static void OnUpdate()
	{
		try
		{
			if (g.world != null && g.world.playerUnit != null)
			{
				if (!_announced)
				{
					_announced = true;
					_runner.Announce();
				}
				_runner.OnUpdate();
			}
		}
		catch (Exception ex)
		{
			Log("OnUpdate 异常: " + ex);
		}
	}

	public static void Shutdown()
	{
		try
		{
			if (_cor != null)
			{
				g.timer.Stop(_cor);
			}
		}
		catch
		{
		}
		_cor = null;
		if (_runner != null)
		{
			_runner.Close();
			_runner = null;
		}
		_announced = false;
	}

	private static void Log(string msg)
	{
		Debug.Log(Object.op_Implicit("[AgentLoopVerify] " + msg));
	}
}

		string text = ((num < _tests.Count) ? string.Format("下一项: {0} {1}", _tests[num].Id, _tests[num].Tool ?? "custom") : "已到末尾（再F8出汇总）");
		Tip($"[验证] 进度 {_cursor + 1}/{_tests.Count} 通过{_passed} 失败{_failed} | {text}");
	}

	private void OpenFile()
	{
		try
		{
			Directory.CreateDirectory(VerifyConfig.ResultDir);
			_file = Path.Combine(VerifyConfig.ResultDir, "verify_" + DateTime.Now.ToString("yyyyMMdd_HHmmss") + ".jsonl");
		}
		catch (Exception ex)
		{
			_file = Path.Combine(Environment.CurrentDirectory, "AgentLoopVerify_fallback.jsonl");
			Log("结果目录不可用(" + ex.Message + ")，改用: " + _file);
		}
	}

	private void AppendLine(string line)
	{
		try
		{
			File.AppendAllText(_file, line + Environment.NewLine, Encoding.UTF8);
		}
		catch (Exception ex)
		{
			Log("结果写入失败: " + ex.Message);
		}
		Log(line);
	}

	private static void Tip(string msg)
	{
		try
		{
			UITipItem.AddTip(msg, 0f);
		}
		catch
		{
		}
		Log(msg);
	}

	private static void Log(string msg)
	{
		try
		{
			Debug.Log(Object.op_Implicit("[AgentLoopVerify] " + msg));
		}
		catch
		{
		}
	}

	private static string Truncate(string s, int len)
	{
		if (string.IsNullOrEmpty(s) || s.Length <= len)
		{
			return s;
		}
		return s.Substring(0, len) + "…";
	}
}
