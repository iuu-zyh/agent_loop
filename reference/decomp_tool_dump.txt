	}

	private static string AreaLabel(string raw)
	{
		if (string.IsNullOrEmpty(raw))
		{
			return raw;
		}
		string text = LSOrRaw(raw);
		if (!string.IsNullOrEmpty(text) && !text.StartsWith("areaName", StringComparison.Ordinal))
		{
			return text;
		}
		if (AreaNameFallback.TryGetValue(raw, out var value))
		{
			return value;
		}
		return string.IsNullOrEmpty(text) ? raw : text;
	}

	private static bool RelationMatch(string relCn, string relation)
	{
		switch (relation)
		{
		case "亲族":
		{
			int result;
			switch (relCn)
			{
			default:
				result = ((relCn == "私生子女") ? 1 : 0);
				break;
			case "父母":
			case "子女":
			case "兄弟姐妹":
			case "义父母":
			case "义子女":
				result = 1;
				break;
			}
			return (byte)result != 0;
		}
		case "道侣":
			return relCn == "道侣" || relCn == "情人";
		case "师徒":
			return relCn == "师傅" || relCn == "徒弟";
		case "结义":
			return relCn == "结义";
		case "好友":
		case "好感":
			return true;
		case "陌生":
			return relCn == "None" || relCn == "";
		default:
			return true;
		}
	}

	{

		BuildInfo buildInfo4 = null;
		foreach (BuildInfo item in CollectNamedBuilds())
		{
			bool flag = string.IsNullOrEmpty(region) || (item.Region != null && item.Region.Contains(region));
			if (item.Name == dest)
			{
				if (flag)
				{
					if (buildInfo3 == null)
					{
						buildInfo3 = item;
					}
				}
				else if (buildInfo == null)
				{
					buildInfo = item;
				}
			}
			else
			{
				if (!item.Name.Contains(dest) && !dest.Contains(item.Name))
				{
					continue;
				}
				if (flag)
				{
					if (buildInfo4 == null)
					{
						buildInfo4 = item;
					}
				}
				else if (buildInfo2 == null)
				{
					buildInfo2 = item;
				}
			}
		}
		return buildInfo3 ?? buildInfo ?? buildInfo4 ?? buildInfo2;
	}
